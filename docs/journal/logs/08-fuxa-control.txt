[2] Modbus 주소 체계 — 드라이버 소스에서

$ curl -sL -o mb.js \
    "https://raw.githubusercontent.com/frangoteam/FUXA/master/server/runtime/devices/modbus/index.js"
$ grep -n "address) - 1" mb.js

216:  var offset = parseInt(data.tags[id].address) - 1;
      // because settings address from 1 to 65536 but communication start from 0

→ FUXA Modbus 주소는 1-based

$ grep -n "ModbusMemoryAddress\s*=" mb.js
842: const ModbusMemoryAddress = { CoilStatus: 0, DigitalInputs: 100000,
                                   InputRegisters: 300000, HoldingRegisters: 400000 };

$ sed -n '/getMemoryAddress = function/,/^    }/p' mb.js
    if (address < ModbusMemoryAddress.DigitalInputs) {
        if (tokenized) return formatAddress('000000', token);   ← 리터럴 '000000'

→ 코일 영역의 memaddress 는 "0" 이 아니라 "000000"

$ curl -sL ".../modbus/datatypes.js" | grep -n "Float32"
83:  Float32: _gen(4, 'FloatBE', 2),          ← 스왑 없음 = big-endian ABCD
117: Float32LE: _gen(4, 'FloatLE', 2),
138: Float32MLE: _gen(4, 'FloatBE', 2, 2),    ← 워드 스왑

→ 시뮬레이터가 struct.pack('>f') 로 쓰므로 Float32 가 맞음

$ docker exec fuxa sed -n '397,405p' /usr/src/app/FUXA/server/runtime/devices/index.js
function getDeviceIdFromTag(sigid) {
    for (var id in activeDevices) {
        var tag = activeDevices[id].getTagProperty(sigid);   ← data.tags[sigid] 직접 조회
        if (tag) return id;
    }
    return null;
}
